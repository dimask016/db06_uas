🗄️ Logistik Inventory Database

✅ FULL DDL + DML (max 10 baris per tabel) | 21 tabel | JOIN product, user, transfer_header, transfer_detail, timlogistik

Halaman ini menampilkan seluruh perintah CREATE TABLE dan seluruh data INSERT dari file logistik_inventory (2).sql (MariaDB/MySQL). Setiap tabel INSERT hanya menampilkan maksimal 10 baris data agar tampilan ringkas, namun seluruh baris asli dapat dilihat di file SQL. Dilengkapi dengan contoh JOIN antara product, user, transfer_header, transfer_detail, dan timlogistik.

📌 1. Semua Perintah CREATE TABLE

Struktur setiap tabel sesuai dump asli (primary key, foreign key, indeks).

🏷️ brand
CREATE TABLE `brand` (
  `brand_id` int(11) NOT NULL,
  `brand_code` varchar(20) NOT NULL,
  `brand_name` varchar(100) NOT NULL,
  `logo_url` varchar(500) DEFAULT NULL,
  `is_active` tinyint(1) DEFAULT 1,
  PRIMARY KEY (`brand_id`),
  UNIQUE KEY `brand_code` (`brand_code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8;
📁 category
CREATE TABLE `category` (
  `category_id` int(11) NOT NULL,
  `category_code` varchar(20) NOT NULL,
  `category_name` varchar(100) NOT NULL,
  `description` text DEFAULT NULL,
  `is_active` tinyint(1) DEFAULT 1,
  PRIMARY KEY (`category_id`),
  UNIQUE KEY `category_code` (`category_code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8;
🚚 driver
CREATE TABLE `driver` (
  `driver_id` int(11) NOT NULL,
  `driver_code` varchar(20) NOT NULL,
  `full_name` varchar(100) NOT NULL,
  `license_number` varchar(30) NOT NULL,
  `license_type` enum('A','B','C','SIM_A','SIM_B','SIM_C') NOT NULL,
  `license_expiry` date NOT NULL,
  `phone` varchar(20) DEFAULT NULL,
  `status` enum('available','on_delivery','off_duty') DEFAULT 'available',
  `joined_date` date DEFAULT NULL,
  `is_active` tinyint(1) DEFAULT 1,
  PRIMARY KEY (`driver_id`),
  UNIQUE KEY `driver_code` (`driver_code`),
  UNIQUE KEY `license_number` (`license_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8;
📦 goodsreceipt
CREATE TABLE `goodsreceipt` (
  `gr_id` int(11) NOT NULL,
  `gr_number` varchar(50) NOT NULL,
  `po_id` int(11) NOT NULL,
  `location_id` int(11) NOT NULL,
  `receipt_date` date NOT NULL,
  `invoice_number` varchar(50) DEFAULT NULL,
  `received_by` int(11) NOT NULL,
  `status` enum('draft','completed','cancelled') DEFAULT 'draft',
  `notes` text DEFAULT NULL,
  PRIMARY KEY (`gr_id`),
  UNIQUE KEY `gr_number` (`gr_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8;
📄 goodsreceiptdetail
CREATE TABLE `goodsreceiptdetail` (
  `gr_detail_id` int(11) NOT NULL,
  `gr_id` int(11) NOT NULL,
  `po_detail_id` int(11) NOT NULL,
  `batch_number` varchar(50) DEFAULT NULL,
  `expiry_date` date DEFAULT NULL,
  `quantity_received` int(11) NOT NULL,
  `quantity_good` int(11) NOT NULL,
  `quantity_damaged` int(11) DEFAULT 0,
  `notes` text DEFAULT NULL,
  PRIMARY KEY (`gr_detail_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8;
📊 inventory
CREATE TABLE `inventory` (
  `inventory_id` int(11) NOT NULL,
  `inventory_code` varchar(20) NOT NULL,
  `product_id` int(11) NOT NULL,
  `location_id` int(11) NOT NULL,
  `quantity_on_rack` int(11) NOT NULL DEFAULT 0,
  `quantity_on_store` int(11) NOT NULL DEFAULT 0,
  `total_stock` int(11) GENERATED ALWAYS AS (`quantity_on_rack` + `quantity_on_store`) STORED,
  `min_stock_level` int(11) DEFAULT 10,
  `last_counted_date` date DEFAULT NULL,
  PRIMARY KEY (`inventory_id`),
  UNIQUE KEY `inventory_code` (`inventory_code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8;
📍 location
CREATE TABLE `location` (
  `location_id` int(11) NOT NULL,
  `location_code` varchar(20) NOT NULL,
  `location_name` varchar(100) NOT NULL,
  `location_type` enum('warehouse','site') NOT NULL,
  `address` text DEFAULT NULL,
  `city` varchar(50) DEFAULT NULL,
  `province` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`location_id`),
  UNIQUE KEY `location_code` (`location_code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8;
🖥️ product
CREATE TABLE `product` (
  `product_id` int(11) NOT NULL,
  `sku` varchar(50) NOT NULL,
  `barcode` varchar(50) DEFAULT NULL,
  `product_name` varchar(200) NOT NULL,
  `description` text DEFAULT NULL,
  `category_id` int(11) NOT NULL,
  `brand_id` int(11) NOT NULL,
  `uom_id` int(11) NOT NULL,
  `weight_kg` decimal(10,2) DEFAULT NULL,
  `is_active` tinyint(1) DEFAULT 1,
  PRIMARY KEY (`product_id`),
  UNIQUE KEY `sku` (`sku`),
  UNIQUE KEY `barcode` (`barcode`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8;
📑 purchaseorder
CREATE TABLE `purchaseorder` (
  `po_id` int(11) NOT NULL,
  `po_number` varchar(50) NOT NULL,
  `supplier_id` int(11) NOT NULL,
  `location_id` int(11) NOT NULL,
  `order_date` date NOT NULL,
  `status` enum('draft','submitted','approved','sent','partial_received','received','cancelled') DEFAULT 'draft',
  `total_amount` decimal(14,2) DEFAULT 0.00,
  `created_by` int(11) NOT NULL,
  PRIMARY KEY (`po_id`),
  UNIQUE KEY `po_number` (`po_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8;
🔖 purchaseorderdetail
CREATE TABLE `purchaseorderdetail` (
  `po_detail_id` int(11) NOT NULL,
  `po_id` int(11) NOT NULL,

  `quantity_ordered` int(11) NOT NULL,

  `discount_percent` decimal(5,2) DEFAULT 0.00,
  PRIMARY KEY (`po_detail_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8;

CREATE TABLE `role` (

  PRIMARY KEY (`role_id`),
  UNIQUE KEY `role_code` (`role_code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8;
🚛 shipment
CREATE TABLE `shipment` (
  `shipment_id` int(11) NOT NULL,
  `location_id` int(11) NOT NULL,

  `shipped_by` int(11) DEFAULT NULL,
  `received_by` int(11) DEFAULT NULL,
  `vehicle_id` int(11) DEFAULT NULL,
  `driver_id` int(11) DEFAULT NULL,

  `status` enum('pending','ready','in_transit','delivered','cancelled') DEFAULT 'pending',
  PRIMARY KEY (`shipment_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8;
📈 stockmovement
CREATE TABLE `stockmovement` (
  `movement_id` int(11) NOT NULL,
  `product_id` int(11) NOT NULL,
  `location_id` int(11) NOT NULL,
  `quantity_change` int(11) NOT NULL,
  `previous_quantity` int(11) NOT NULL,
  `new_quantity` int(11) NOT NULL,
  `movement_type` enum('IN_purchase','IN_transfer','IN_return','IN_adjustment_plus','OUT_sale','OUT_transfer','OUT_damage','OUT_expired','OUT_adjustment_minus','OUT_usage') NOT NULL,
  `reference_type` varchar(50) DEFAULT NULL,
  `reference_id` int(11) DEFAULT NULL,
  `created_by` int(11) NOT NULL,
  PRIMARY KEY (`movement_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8;
🏢 supplier
CREATE TABLE `supplier` (
  `supplier_id` int(11) NOT NULL,
  `supplier_code` varchar(20) NOT NULL,
  `supplier_name` varchar(150) NOT NULL,
  `contact_person` varchar(100) DEFAULT NULL,
  `supplier_email` varchar(100) DEFAULT NULL,
  `supplier_phone` varchar(20) DEFAULT NULL,
  `supplier_address` text DEFAULT NULL,
  PRIMARY KEY (`supplier_id`),
  UNIQUE KEY `supplier_code` (`supplier_code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8;
👥 timlogistik
CREATE TABLE `timlogistik` (
  `logistik_id` int(11) NOT NULL,
  `logistik_name` varchar(30) NOT NULL,
  `location_id` int(11) NOT NULL,
  `position` enum('supervisor','coordinator','staff','Manager') NOT NULL,
  `shift` enum('morning','afternoon','night','nonshift') NOT NULL,
  `join_date` date NOT NULL,
  `is_active` tinyint(1) DEFAULT 1,
  PRIMARY KEY (`logistik_id`),
  UNIQUE KEY `logistik_name` (`logistik_name`),
  CONSTRAINT `timlogistik_ibfk_1` FOREIGN KEY (`logistik_name`) REFERENCES `user` (`username`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8;
📋 transferrequest
CREATE TABLE `transferrequest` (
  `transfer_request_id` int(11) NOT NULL,
  `request_number` varchar(50) NOT NULL,
  `product_id` int(11) NOT NULL,
  `location_id` int(11) DEFAULT NULL,
  `requested_quantity` int(11) NOT NULL,
  `status` enum('pending','approved','rejected','cancelled','fulfilled') DEFAULT 'pending',
  `requested_by` int(11) NOT NULL,
  `approved_by` int(11) DEFAULT NULL,
  PRIMARY KEY (`transfer_request_id`),
  UNIQUE KEY `request_number` (`request_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8;
📦 transfer_detail
CREATE TABLE `transfer_detail` (
  `transfer_detail_id` int(11) NOT NULL,
  `transfer_id` int(11) NOT NULL,
  `product_id` int(11) NOT NULL,
  `quantity` int(11) NOT NULL,
  `quantity_shipped` int(11) DEFAULT 0,
  `quantity_received` int(11) DEFAULT 0,
  `status` enum('pending','shipped','received','cancelled') DEFAULT 'pending',
  `notes` text DEFAULT NULL,
  PRIMARY KEY (`transfer_detail_id`),
  KEY `idx_transfer_id` (`transfer_id`),
  CONSTRAINT `transfer_detail_ibfk_1` FOREIGN KEY (`transfer_id`) REFERENCES `transfer_header` (`transfer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8;
🚚 transfer_header
CREATE TABLE `transfer_header` (
  `transfer_id` int(11) NOT NULL,
  `transfer_number` varchar(50) NOT NULL,
  `location_id` int(11) NOT NULL,
  `logistik_id` int(11) DEFAULT NULL,
  `transfer_date` timestamp NULL DEFAULT current_timestamp(),
  `status` enum('draft','pending','picking','shipped','received','cancelled') DEFAULT 'draft',
  `received_date` timestamp NULL DEFAULT NULL,
  `received_by` int(11) DEFAULT NULL,
  `created_by` int(11) NOT NULL,
  PRIMARY KEY (`transfer_id`),
  UNIQUE KEY `transfer_number` (`transfer_number`),
  KEY `fk_transfer_header_logistik` (`logistik_id`),
  CONSTRAINT `fk_transfer_header_logistik` FOREIGN KEY (`logistik_id`) REFERENCES `timlogistik` (`logistik_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8;
📏 unitofmeasure
CREATE TABLE `unitofmeasure` (
  `uom_id` int(11) NOT NULL,
  `uom_code` varchar(10) NOT NULL,
  `uom_name` varchar(50) NOT NULL,
  `conversion_factor` decimal(10,4) DEFAULT 1.0000,
  `is_base` tinyint(1) DEFAULT 0,
  PRIMARY KEY (`uom_id`),
  UNIQUE KEY `uom_code` (`uom_code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8;
👤 user
CREATE TABLE `user` (
  `user_id` int(11) NOT NULL,
  `username` varchar(50) NOT NULL,
  `password_hash` varchar(255) NOT NULL,
  `full_name` varchar(100) NOT NULL,
  `email` varchar(100) NOT NULL,
  `phone` varchar(20) DEFAULT NULL,
  `role_id` int(11) NOT NULL,
  `location_id` int(11) DEFAULT NULL,
  `is_active` tinyint(1) DEFAULT 1,
  PRIMARY KEY (`user_id`),
  UNIQUE KEY `uk_username` (`username`),
  UNIQUE KEY `uk_email` (`email`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8;
🚐 vehicle
CREATE TABLE `vehicle` (
  `vehicle_id` int(11) NOT NULL,
  `vehicle_code` varchar(20) NOT NULL,
  `plate_number` varchar(20) NOT NULL,
  `vehicle_type` enum('truck','van','motorcycle') NOT NULL,
  `capacity_kg` int(11) DEFAULT NULL,
  `status` enum('available','on_delivery','maintenance') DEFAULT 'available',
  PRIMARY KEY (`vehicle_id`),
  UNIQUE KEY `vehicle_code` (`vehicle_code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8;

🔗 3. JOIN tabel product, user, transfer_header, transfer_detail, timlogistik

🎯 Relasi kunci :
• transfer_detail ↔ transfer_header (transfer_id)
• transfer_detail.product_id ↔ product
• transfer_header.created_by ↔ user.user_id (pembuat)
• transfer_header.logistik_id ↔ timlogistik.logistik_id (koordinator)
• timlogistik.logistik_name = user.username (opsional)

📌 Query JOIN lengkap (5 tabel)

SELECT 
    th.transfer_number,
    p.product_name,
    td.quantity AS qty_dikirim,
    creator.full_name AS dibuat_oleh,
    tl.logistik_name AS koordinator_logistik,
    th.status
FROM transfer_header th
JOIN transfer_detail td ON th.transfer_id = td.transfer_id
JOIN product p ON td.product_id = p.product_id
LEFT JOIN user creator ON th.created_by = creator.user_id
LEFT JOIN timlogistik tl ON th.logistik_id = tl.logistik_id
ORDER BY th.transfer_date DESC;
